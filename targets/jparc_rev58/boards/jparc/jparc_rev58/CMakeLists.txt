# Apply the VOS1 adaptation only to this board, without editing the workspace SDK.
function(jparc_rev58_clock_vos1)
  set(upstream "${ZEPHYR_BASE}/drivers/clock_control/clock_stm32_ll_h7.c")
  file(SHA256 "${upstream}" actual_hash)
  if(NOT actual_hash STREQUAL "c97fbba7b94b236fec153d404a0e78b094dd03b400da454be32def140547df66")
    message(FATAL_ERROR "rev58 VOS1 adaptation: Zephyr clock driver changed; review power initialization before updating the hash")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${upstream}")
  file(READ "${upstream}" source)
  string(REPLACE "PWR_REGULATOR_VOLTAGE_SCALE0" "PWR_REGULATOR_VOLTAGE_SCALE1" source "${source}")
  string(REPLACE "LL_PWR_REGU_VOLTAGE_SCALE0" "LL_PWR_REGU_VOLTAGE_SCALE1" source "${source}")
  string(REPLACE "Highest voltage is SCALE0" "rev58 Direct SMPS requires SCALE1" source "${source}")
  set(patched "${CMAKE_BINARY_DIR}/jparc_rev58/clock_stm32_ll_h7_vos1.c")
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/jparc_rev58")
  file(WRITE "${patched}" "${source}")
  get_target_property(sources drivers__clock_control SOURCES)
  set(found FALSE)
  set(updated)
  foreach(src IN LISTS sources)
    if(src MATCHES "(^|/)clock_stm32_ll_h7[.]c$")
      list(APPEND updated "${patched}")
      set(found TRUE)
    else()
      list(APPEND updated "${src}")
    endif()
  endforeach()
  if(NOT found)
    message(FATAL_ERROR "rev58 VOS1 adaptation: STM32H7 clock source was not found")
  endif()
  set_property(TARGET drivers__clock_control PROPERTY SOURCES "${updated}")
  target_include_directories(drivers__clock_control PRIVATE "${ZEPHYR_BASE}/drivers/clock_control")
  message(STATUS "rev58: using board-local VOS1 clock driver")
endfunction()
# Drivers are added after boards; replace the source after that target exists.
cmake_language(DEFER DIRECTORY "${ZEPHYR_BASE}" CALL jparc_rev58_clock_vos1)
zephyr_library()
zephyr_library_sources(board_checks.c)

# Pinned LAN8742 init treats an already-correct advertisement as a fatal error.
# Accept that result only during init; retain all other PHY error handling.
function(jparc_rev58_phy_already_configured)
  if(NOT CONFIG_PHY_MICROCHIP_LAN8742)
    return()
  endif()
  set(upstream "${ZEPHYR_BASE}/drivers/ethernet/phy/phy_microchip_lan8742.c")
  file(SHA256 "${upstream}" actual_hash)
  if(NOT actual_hash STREQUAL "5512da25a0da3df610f0d388463b7b757ef20ceb74d78a2276776a2729924734")
    message(FATAL_ERROR "rev58 LAN8742 adaptation: upstream changed; review EALREADY handling before updating the hash")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${upstream}")
  file(READ "${upstream}" source)
  set(before "ret = phy_lan8742_cfg_link(dev, cfg->default_speeds, 0);\n\tif (ret < 0) {")
  set(after "ret = phy_lan8742_cfg_link(dev, cfg->default_speeds, 0);\n\t/* rev58: unchanged settings are a successful initialization. */\n\tif (ret < 0 && ret != -EALREADY) {")
  string(FIND "${source}" "${before}" match)
  if(match EQUAL -1)
    message(FATAL_ERROR "rev58 LAN8742 adaptation: init pattern not found")
  endif()
  string(REPLACE "${before}" "${after}" source "${source}")
  set(patched "${CMAKE_BINARY_DIR}/jparc_rev58/phy_microchip_lan8742_init.c")
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/jparc_rev58")
  file(WRITE "${patched}" "${source}")
  get_target_property(sources drivers__ethernet SOURCES)
  set(found FALSE)
  set(updated)
  foreach(src IN LISTS sources)
    if(src MATCHES "(^|/)phy_microchip_lan8742[.]c$")
      list(APPEND updated "${patched}")
      set(found TRUE)
    else()
      list(APPEND updated "${src}")
    endif()
  endforeach()
  if(NOT found)
    message(FATAL_ERROR "rev58 LAN8742 adaptation: PHY source was not found")
  endif()
  set_property(TARGET drivers__ethernet PROPERTY SOURCES "${updated}")
  target_include_directories(drivers__ethernet PRIVATE "${ZEPHYR_BASE}/drivers/ethernet/phy")
  message(STATUS "rev58: accepting LAN8742 already-configured result at init")
endfunction()
cmake_language(DEFER DIRECTORY "${ZEPHYR_BASE}" CALL jparc_rev58_phy_already_configured)
